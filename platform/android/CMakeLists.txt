# ── navaLauncher (Android) ───────────────────────────────────────────────────
#
# Driven by app/build.gradle's externalNativeBuild. Produces libnava_launcher.so:
# the app, app_shell's Android host, vk_canvas core, and the NDK's native_app_glue.
cmake_minimum_required(VERSION 3.22.1)
project(nava_launcher_android CXX C)

set(REPO_ROOT     ${CMAKE_CURRENT_SOURCE_DIR}/../..)
set(NAVA_ROOT     ${REPO_ROOT})
set(FRAMEWORK_DIR ${REPO_ROOT}/framework)
set(VK_CANVAS_DIR ${FRAMEWORK_DIR}/Vk_Canvas_Lb_LAW CACHE PATH "" FORCE)

set(APP_SHELL_APP_NAME "navalauncher" CACHE STRING "" FORCE)

# native_app_glue ships inside the NDK, so only a consumer knows where it is.
set(NATIVE_APP_GLUE ${ANDROID_NDK}/sources/android/native_app_glue)
set(APP_SHELL_NATIVE_APP_GLUE_DIR ${NATIVE_APP_GLUE} CACHE PATH "" FORCE)

add_subdirectory(${VK_CANVAS_DIR}/core      ${CMAKE_BINARY_DIR}/vk_canvas_core)
add_subdirectory(${FRAMEWORK_DIR}/App_shell ${CMAKE_BINARY_DIR}/app_shell)

# Shaders and font land in the Gradle module's assets/, which AGP packs into
# the APK. See cmake/NavaAssets.cmake on why they are copied rather than
# compiled.
list(APPEND CMAKE_MODULE_PATH ${REPO_ROOT}/cmake)
include(NavaAssets)

find_program(VCE_SLANGC slangc HINTS /opt/shader-slang/bin $ENV{VULKAN_SDK}/bin REQUIRED)

# The MSDF text-size floor is generated by a tool that has to RUN on the build
# machine, so it cannot be built here. A desktop configure produces it; this
# build reads it out of that tree. Point NAVA_GENERATED_DIR at
# <desktop-build>/generated if it is not the default below.
set(NAVA_GENERATED_DIR ${REPO_ROOT}/build/generated CACHE PATH
    "Where a desktop build left ui_min_text_size.gen.h")

# app_shell's ui_metrics.hh includes it and looks in ${CMAKE_BINARY_DIR}/generated,
# so the copy has to land there — an include path on our own target would not
# help the library that does the including.
if(NOT EXISTS ${NAVA_GENERATED_DIR}/ui_min_text_size.gen.h)
    message(FATAL_ERROR
        "ui_min_text_size.gen.h is missing. It is generated by a tool that has "
        "to RUN on the build machine, so the Android build cannot make it: "
        "configure the desktop build first (cmake -S . -B build), or point "
        "-DNAVA_GENERATED_DIR at a tree that has it.")
endif()
configure_file(${NAVA_GENERATED_DIR}/ui_min_text_size.gen.h
               ${CMAKE_BINARY_DIR}/generated/ui_min_text_size.gen.h COPYONLY)

add_library(nava_launcher SHARED
    ${REPO_ROOT}/app/android_main.cc
    ${REPO_ROOT}/app/launcher_app.cc
    ${REPO_ROOT}/app/launcher_draw.cc
    ${REPO_ROOT}/app/app_list.cc
    ${REPO_ROOT}/app/app_cache.cc
    ${REPO_ROOT}/os/pm_bridge.cc
    ${NATIVE_APP_GLUE}/android_native_app_glue.c
    # vk_canvas's Android platform seam: AndroidSurfaceProvider and
    # AndroidAssetReader, which AndroidHost constructs. It is a .cc a consumer
    # compiles rather than a library, the same way the Win32 backend is.
    ${VK_CANVAS_DIR}/platform/android/android_platform.cc)

target_include_directories(nava_launcher PRIVATE
    ${REPO_ROOT}/app
    ${REPO_ROOT}/os
    ${NATIVE_APP_GLUE}
    ${VK_CANVAS_DIR}/platform/android   # jni_util.hh
    ${NAVA_GENERATED_DIR})

target_compile_options(nava_launcher PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-std=c++17> -O2)

target_link_libraries(nava_launcher
    app_shell_android
    vk_canvas_core
    android log vulkan)

# The glue's entry point has no caller in this translation unit; without this
# the linker garbage-collects the symbol NativeActivity looks up by name.
target_link_options(nava_launcher PRIVATE -u ANativeActivity_onCreate)

nava_stage_assets(nava_launcher ${CMAKE_CURRENT_SOURCE_DIR}/app/src/main/assets)
